### 4 个关键决策

| # | 决策 | 选择 | 理由 |
|---|---|---|---|
| D1 | 多格式库 | python-docx + python-pptx + openpyxl | 轻、专一、各 ~5MB；unstructured 200MB+ 还要 LibreOffice |
| D2 | docx/xlsx 的 page_num | docx=None，xlsx=sheet index，pptx=slide num | "page" 是 PDF/PPT 概念；其他格式语义不同，前端按 metadata 显示 |
| D3 | /chat/stream 与 /chat 关系 | 共存。/chat 保留同步版作 fallback；前端默认走 /chat/stream | 流式失败可降级；不破坏向后兼容 |
| D4 | Rerank 后端 | Cohere API（默认）；无 key fallback 到纯 cosine | 0 容器膨胀；切本地 BGE 是 M3 事 |

### Kill criteria（任一触发 → 砍单功能，剩下继续）

- **多格式 P0**：unstructured-fallback 报错 / xlsx 解析空 → 砍 xlsx，保 docx+pptx
- **流式 P1**：浏览器 SSE 与 Anthropic 流格式接不上超过 4h → 退回同步版
- **重排 P2**：Cohere API HK 调不通 / Key 申请不到 → 跳过，纯 cosine 走

```python
from io import BytesIO
from pathlib import Path

from pypdf import PdfReader


def parse_pdf(source) -> list[dict]:
    """Returns list of {page_num, text, kind: 'page'}."""
    if isinstance(source, (bytes, bytearray)):
        reader = PdfReader(BytesIO(source))
    else:
        reader = PdfReader(str(source))

    pages: list[dict] = []
    for i, page in enumerate(reader.pages, start=1):
        text = (page.extract_text() or "").strip()
        if text:
            pages.append({"page_num": i, "text": text, "kind": "page"})
    return pages
```

```python
from pathlib import Path

from app.services.parsers import parse_docx, parse_pdf, parse_pptx, parse_xlsx


def parse(source: bytes | str | Path, mime_type: str | None = None, filename: str | None = None) -> list[dict]:
    """统一入口。返回 [{page_num, text, kind}]。

    page_num 语义：
      - pdf: 1-based 页码
      - pptx: 1-based 幻灯片号
      - xlsx: sheet index (1-based)
      - docx: None（按段落组分块，无"页"概念）
    """
    ext = _detect(mime_type, filename)
    if ext == "pdf":
        return parse_pdf(source)
    if ext == "docx":
        return parse_docx(source)
    if ext == "pptx":
        return parse_pptx(source)
    if ext == "xlsx":
        return parse_xlsx(source)
    raise ValueError(f"Unsupported file type: mime={mime_type} filename={filename}")


_EXT_MAP = {
    "application/pdf": "pdf",
    "application/vnd.openxmlformats-officedocument.wordprocessingml.document": "docx",
    "application/vnd.openxmlformats-officedocument.presentationml.presentation": "pptx",
    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet": "xlsx",
}


def _detect(mime_type: str | None, filename: str | None) -> str:
    if mime_type and mime_type in _EXT_MAP:
        return _EXT_MAP[mime_type]
    if filename:
        suffix = Path(filename).suffix.lower().lstrip(".")
        if suffix in {"pdf", "docx", "pptx", "xlsx"}:
            return suffix
    raise ValueError("cannot detect file type")
```

```python
# rag-backend/app/services/parsers/docx.py
from io import BytesIO
from pathlib import Path

from docx import Document as _Doc


_TARGET_CHARS = 2000  # 每段最大字符数，超过分新 section


def parse_docx(source) -> list[dict]:
    """Returns list of {page_num=None, text, kind='section'}.

    docx 无页概念，按段落聚合到 ~2000 字符的 section。
    """
    if isinstance(source, (bytes, bytearray)):
        doc = _Doc(BytesIO(source))
    else:
        doc = _Doc(str(source))

    paragraphs = [p.text.strip() for p in doc.paragraphs if p.text and p.text.strip()]
    sections: list[dict] = []
    buf: list[str] = []
    buf_len = 0

    for para in paragraphs:
        if buf_len + len(para) > _TARGET_CHARS and buf:
            sections.append({"page_num": None, "text": "\n".join(buf), "kind": "section"})
            buf = []
            buf_len = 0
        buf.append(para)
        buf_len += len(para)

    if buf:
        sections.append({"page_num": None, "text": "\n".join(buf), "kind": "section"})

    return sections
```

```python
# rag-backend/app/services/parsers/pptx.py
from io import BytesIO

from pptx import Presentation


def parse_pptx(source) -> list[dict]:
    """Returns list of {page_num=<slide#>, text, kind='slide'}.

    每个 slide 一个 chunk；包括标题、内容、备注。
    """
    if isinstance(source, (bytes, bytearray)):
        prs = Presentation(BytesIO(source))
    else:
        prs = Presentation(str(source))

    chunks: list[dict] = []
    for i, slide in enumerate(prs.slides, start=1):
        parts: list[str] = []
        for shape in slide.shapes:
            if shape.has_text_frame:
                for para in shape.text_frame.paragraphs:
                    line = "".join(run.text for run in para.runs).strip()
                    if line:
                        parts.append(line)
        # speaker notes
        if slide.has_notes_slide:
            notes_text = (slide.notes_slide.notes_text_frame.text or "").strip()
            if notes_text:
                parts.append(f"[备注] {notes_text}")
        text = "\n".join(parts).strip()
        if text:
            chunks.append({"page_num": i, "text": text, "kind": "slide"})
    return chunks
```

```python
# rag-backend/app/services/parsers/xlsx.py
from io import BytesIO

from openpyxl import load_workbook


def parse_xlsx(source) -> list[dict]:
    """Returns list of {page_num=<sheet_idx>, text, kind='sheet'}.

    每个 sheet 一个 chunk，rows 用 ' | ' 分隔，列用 ' \\t ' 分隔。
    """
    if isinstance(source, (bytes, bytearray)):
        wb = load_workbook(filename=BytesIO(source), data_only=True, read_only=True)
    else:
        wb = load_workbook(filename=str(source), data_only=True, read_only=True)

    chunks: list[dict] = []
    for i, sheet_name in enumerate(wb.sheetnames, start=1):
        ws = wb[sheet_name]
        rows_text: list[str] = []
        for row in ws.iter_rows(values_only=True):
            cells = [str(c) if c is not None else "" for c in row]
            if any(c.strip() for c in cells):
                rows_text.append("\t".join(cells))
        body = " | ".join(rows_text)
        if body.strip():
            chunks.append({
                "page_num": i,
                "text": f"[Sheet: {sheet_name}]\n{body}",
                "kind": "sheet",
            })
    wb.close()
    return chunks
```

```python
from typing import AsyncIterator, Iterator


def generate_answer_stream(
    query: str, chunks: list[RetrievedChunk]
) -> Iterator[tuple[str, str | list[Citation]]]:
    """生成器，yield (event_type, payload)。

    event types:
      - "text": payload=str (片段)
      - "citations": payload=list[Citation] (流末)
      - "error": payload=str
    """
    if not chunks:
        yield ("text", "根据现有知识库无法回答这个问题。")
        yield ("citations", [])
        return

    settings = get_settings()
    user_prompt = _build_user_prompt(query, chunks)

    with _client().messages.stream(
        model=settings.anthropic_model,
        max_tokens=1024,
        system=SYSTEM_PROMPT,
        messages=[{"role": "user", "content": user_prompt}],
    ) as stream:
        full_text = ""
        for delta in stream.text_stream:
            full_text += delta
            yield ("text", delta)

    # 流结束，解析 citations
    citation_indices = extract_citations(full_text)
    citations: list[Citation] = []
    for idx in citation_indices:
        if 1 <= idx <= len(chunks):
            ck = chunks[idx - 1]
            citations.append(Citation(
                doc_id=ck.doc_id, filename=ck.filename, page_num=ck.page_num,
                chunk_id=ck.chunk_id, score=ck.score,
            ))
    yield ("citations", citations)
```

```python
import json
from sse_starlette.sse import EventSourceResponse

from app.services.generation_service import generate_answer_stream


@router.post("/chat/stream")
def chat_stream(req: ChatRequest, db: Session = Depends(get_db)):
    if req.session_id:
        session = db.get(ChatSession, req.session_id)
        if session is None:
            raise HTTPException(404, "session not found")
    else:
        session = ChatSession(title=req.message[:50])
        db.add(session); db.flush()

    db.add(Message(session_id=session.id, role="user", content=req.message))
    db.flush()

    t0 = time.perf_counter()
    retrieved = retrieve(db, req.message)
    retrieval_ms = int((time.perf_counter() - t0) * 1000)

    session_id = session.id
    full_answer_parts: list[str] = []
    final_citations: list[Citation] = []

    def event_gen():
        # 首事件：session info
        yield {"event": "session", "data": json.dumps({"session_id": session_id})}

        t1 = time.perf_counter()
        for event_type, payload in generate_answer_stream(req.message, retrieved):
            if event_type == "text":
                full_answer_parts.append(payload)
                yield {"event": "text", "data": json.dumps({"delta": payload})}
            elif event_type == "citations":
                final_citations.extend(payload)
                yield {
                    "event": "citations",
                    "data": json.dumps({"citations": [c.model_dump() for c in payload]}),
                }
        generation_ms = int((time.perf_counter() - t1) * 1000)

        yield {"event": "done", "data": json.dumps({
            "retrieval_ms": retrieval_ms,
            "generation_ms": generation_ms,
        })}

        # 持久化（流末做）— 用新 session 避免主 db 已关
        from app.dependencies import _SessionLocal
        from app.models import RetrievalLog
        with _SessionLocal() as db2:
            db2.add(Message(
                session_id=session_id,
                role="assistant",
                content="".join(full_answer_parts),
                citations=[c.model_dump() for c in final_citations],
            ))
            db2.add(RetrievalLog(
                session_id=session_id,
                query=req.message,
                retrieved_chunks=[{"chunk_id": r.chunk_id, "doc": r.filename, "page": r.page_num, "score": r.score} for r in retrieved],
                chunks_sent_to_llm=len(retrieved),
                retrieval_latency_ms=retrieval_ms,
                generation_latency_ms=generation_ms,
            ))
            db2.commit()

    return EventSourceResponse(event_gen())
```

```typescript
const send = async () => {
  const trimmed = input.trim();
  if (!trimmed || loading) return;
  setError(null);
  setLoading(true);

  setMessages((m) => [...m, { role: "user", content: trimmed }]);
  setInput("");

  // 占位 assistant message，会被逐字 append
  const assistantIdx = messages.length + 1; // user 已加，此处 idx 对
  setMessages((m) => [...m, { role: "assistant", content: "", citations: [] }]);

  try {
    const res = await fetch(`${API}/chat/stream`, {
      method: "POST",
      headers: { "Content-Type": "application/json", Accept: "text/event-stream" },
      body: JSON.stringify({ session_id: sessionId, message: trimmed }),
    });
    if (!res.ok || !res.body) throw new Error(`HTTP ${res.status}`);

    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buffer = "";

    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });

      // SSE 事件以 \n\n 分割；每事件含 event: X\ndata: {...}
      let idx;
      while ((idx = buffer.indexOf("\n\n")) >= 0) {
        const raw = buffer.slice(0, idx);
        buffer = buffer.slice(idx + 2);
        const evMatch = raw.match(/^event:\s*(.+)$/m);
        const dataMatch = raw.match(/^data:\s*(.+)$/m);
        if (!evMatch || !dataMatch) continue;
        const event = evMatch[1].trim();
        const data = JSON.parse(dataMatch[1]);

        if (event === "session") {
          setSessionId(data.session_id);
        } else if (event === "text") {
          setMessages((m) => {
            const next = [...m];
            next[next.length - 1] = {
              ...next[next.length - 1],
              content: (next[next.length - 1].content || "") + data.delta,
            };
            return next;
          });
        } else if (event === "citations") {
          setMessages((m) => {
            const next = [...m];
            next[next.length - 1] = { ...next[next.length - 1], citations: data.citations };
            return next;
          });
        }
      }
    }
  } catch (e: any) {
    console.error(e);
    setError("服务暂时无响应，请稍后再试。");
  } finally {
    setLoading(false);
  }
};
```

```python
from functools import lru_cache

import cohere

from app.config import get_settings


@lru_cache
def _cohere_client():
    return cohere.ClientV2(api_key=get_settings().cohere_api_key)


def retrieve(db: Session, query: str, top_k: int | None = None) -> list[RetrievedChunk]:
    settings = get_settings()
    top_k = top_k or settings.top_k

    q_vec = embed_query(query)

    # 1. cosine 召回 candidate_k 个
    stmt = (
        select(Chunk.id, Chunk.document_id, Document.filename, Chunk.page_num,
               Chunk.content, (1 - Chunk.embedding.cosine_distance(q_vec)).label("score"))
        .join(Document, Document.id == Chunk.document_id)
        .order_by(Chunk.embedding.cosine_distance(q_vec))
        .limit(settings.retrieval_candidate_k)
    )
    rows = db.execute(stmt).all()
    candidates = [
        RetrievedChunk(
            chunk_id=r.id, doc_id=r.document_id, filename=r.filename,
            page_num=r.page_num, content=r.content, score=float(r.score),
        )
        for r in rows
    ]

    # 2. 重排（有 key 用 Cohere；否则纯 cosine top-k）
    if not settings.cohere_api_key or len(candidates) <= top_k:
        return candidates[:top_k]

    try:
        resp = _cohere_client().rerank(
            model=settings.cohere_rerank_model,
            query=query,
            documents=[c.content for c in candidates],
            top_n=top_k,
        )
        # resp.results = [{index, relevance_score}, ...] 已按分数排序
        reranked: list[RetrievedChunk] = []
        for r in resp.results:
            original = candidates[r.index]
            reranked.append(RetrievedChunk(
                chunk_id=original.chunk_id, doc_id=original.doc_id,
                filename=original.filename, page_num=original.page_num,
                content=original.content,
                score=float(r.relevance_score),  # 重排分数覆盖 cosine
            ))
        return reranked
    except Exception as e:
        import logging
        logging.getLogger(__name__).warning("cohere rerank failed, fallback to cosine: %s", e)
        return candidates[:top_k]
```
